// Orange Pi RV2 boot environment, imported by U-Boot from the bootfs
// partition (see board/spacemit/k1-x/k1x.c: env_${CONFIG_SYS_CONFIG_NAME}.txt)
// devtype/devnum/distro_bootpart are set by U-Boot for the boot device

// Serial console
baudrate=115200
stdin=serial,usbkbd
stdout=serial
stderr=serial

// Load addresses
kernel_addr_r=0x10000000
dtb_addr=0x1F000000

// Kernel 7.2 device tree (from arch/riscv/boot/dts/spacemit/)
dtb_name=dtbs/spacemit/k1-orangepi-rv2.dtb
efi_name=EFI/systemd/systemd-bootriscv64.efi

// NVMe (whole-disk image cloned to NVMe: 1=uboot 2=bootfs 3=rootfs)
nvme_devnum=0
nvme_bootpart=2

efi_boot_nvme=\
           load nvme ${nvme_devnum}:${nvme_bootpart} ${dtb_addr} ${dtb_name}; \
           load nvme ${nvme_devnum}:${nvme_bootpart} ${kernel_addr_r} ${efi_name}; \
           bootefi ${kernel_addr_r} ${dtb_addr};

// MMC bootfs
efi_boot_mmc=\
           load ${devtype} ${devnum}:${distro_bootpart} ${dtb_addr} ${dtb_name}; \
           load ${devtype} ${devnum}:${distro_bootpart} ${kernel_addr_r} ${efi_name}; \
           bootefi ${kernel_addr_r} ${dtb_addr};

// Try NVMe first, fall back to MMC
bootcmd=if nvme scan; then run efi_boot_nvme; fi; run efi_boot_mmc

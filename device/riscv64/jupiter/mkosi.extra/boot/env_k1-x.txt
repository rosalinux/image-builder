// Milk-V Jupiter boot environment, imported by U-Boot from the bootfs
// partition (see board/spacemit/k1-x/k1x.c: env_${CONFIG_SYS_CONFIG_NAME}.txt)
// devtype/devnum/distro_bootpart are set by U-Boot for the boot device
//
// Two boot paths exist on this board:
//  * SD inserted: ROM loads OUR u-boot from SD (SD is always tried first).
//    Our build cannot train the PCIe link (slot power/PHY not described in
//    its DTS), so nvme scan always fails there - do not put nvme commands
//    on the SD path.
//  * No SD: ROM falls back to the SPI NOR bootloader (vendor build, which
//    does bring up PCIe); it imports this file from the NVMe bootfs and
//    boots kernel+rootfs entirely from NVMe.

// Serial console
baudrate=115200
stdin=serial,usbkbd
stdout=serial
stderr=serial

// Load addresses
kernel_addr_r=0x10000000
ramdisk_addr_r=0x21000000
dtb_addr=0x1F000000

// Kernel/initrd/dtb names; @KREL@ is replaced with the actual kernel
// release by mkosi.postinst.chroot
kernel_name=vmlinuz-@KREL@
initrd_name=initramfs-@KREL@.img
dtb_name=dtb-@KREL@/spacemit/k1-milkv-jupiter.dtb
efi_name=EFI/systemd/systemd-bootriscv64.efi

// NVMe (whole-disk image cloned to NVMe: 1=uboot 2=bootfs 3=rootfs).
// Only usable under the NOR (vendor) u-boot - see the note above.
nvme_devnum=0
nvme_bootpart=2

efi_boot_nvme=\
           load nvme ${nvme_devnum}:${nvme_bootpart} ${dtb_addr} ${dtb_name}; \
           load nvme ${nvme_devnum}:${nvme_bootpart} ${kernel_addr_r} ${efi_name}; \
           bootefi ${kernel_addr_r} ${dtb_addr};

// SD bootfs: direct booti (kernel + initramfs + dtb from the FAT bootfs)
boot_mmc=\
           load ${devtype} ${devnum}:${distro_bootpart} ${kernel_addr_r} ${kernel_name}; \
           load ${devtype} ${devnum}:${distro_bootpart} ${ramdisk_addr_r} ${initrd_name}; \
           setenv initrd_size ${filesize}; \
           load ${devtype} ${devnum}:${distro_bootpart} ${dtb_addr} ${dtb_name}; \
           setenv bootargs root=LABEL=rootfs rootwait rw console=ttyS0,115200 loglevel=7; \
           booti ${kernel_addr_r} ${ramdisk_addr_r}:${initrd_size} ${dtb_addr}

bootcmd=if nvme scan; then run efi_boot_nvme; fi; run boot_mmc
